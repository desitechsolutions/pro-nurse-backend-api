### 1. Emergency SOS Feature

#### Overview
Immediate dispatch system for medical emergencies with priority handling and multiple nurse alerts.

#### Database Schema

```sql
-- Migration: V11__add_emergency_sos_support.sql

-- Add emergency flag to bookings
ALTER TABLE bookings ADD COLUMN is_emergency BOOLEAN DEFAULT FALSE;
ALTER TABLE bookings ADD COLUMN emergency_description TEXT;
ALTER TABLE bookings ADD COLUMN emergency_contact_name VARCHAR(100);
ALTER TABLE bookings ADD COLUMN emergency_contact_mobile VARCHAR(15);

-- Create emergency contacts table
CREATE TABLE emergency_contacts (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id BIGINT NOT NULL,
    contact_name VARCHAR(100) NOT NULL,
    contact_mobile VARCHAR(15) NOT NULL,
    relationship VARCHAR(50),
    is_primary BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT fk_emergency_contact_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE INDEX idx_emergency_contacts_user ON emergency_contacts(user_id);
```

### 2. Nurse Availability Calendar

#### Overview
Allows nurses to set their working hours, block specific dates, and manage recurring schedules.

#### Database Schema

```sql
-- Migration: V12__add_nurse_availability_calendar.sql

CREATE TABLE nurse_availability_schedules (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nurse_user_id BIGINT NOT NULL,
    day_of_week INT NOT NULL, -- 1=Monday, 7=Sunday
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT fk_schedule_nurse FOREIGN KEY (nurse_user_id) REFERENCES users(id) ON DELETE CASCADE,
    CONSTRAINT chk_day_of_week CHECK (day_of_week BETWEEN 1 AND 7)
);

CREATE TABLE nurse_blocked_dates (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nurse_user_id BIGINT NOT NULL,
    blocked_date DATE NOT NULL,
    reason VARCHAR(255),
    created_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT fk_blocked_nurse FOREIGN KEY (nurse_user_id) REFERENCES users(id) ON DELETE CASCADE,
    CONSTRAINT uk_nurse_blocked_date UNIQUE (nurse_user_id, blocked_date)
);

CREATE INDEX idx_availability_nurse ON nurse_availability_schedules(nurse_user_id);
CREATE INDEX idx_blocked_dates_nurse ON nurse_blocked_dates(nurse_user_id, blocked_date);
```

#### API Endpoints

**1. Set Weekly Schedule**
```java
POST /api/nurse/availability/schedule

Request:
{
  "schedules": [
    {

    },
    {
      "dayOfWeek": 2,  // Tuesday
      "startTime": "09:00",
      "endTime": "17:00"
    }
  ]
}
```

**2. Block Specific Dates**
```java
POST /api/nurse/availability/block-dates
Request:
{
  "dates": ["2024-12-25", "2024-12-26"],
  "reason": "Holiday vacation"
}
```

**3. Get Availability**
```java
GET /api/nurse/availability
GET /api/nurse/availability/blocked-dates
```

#### Implementation Features
- ✅ Weekly recurring schedule
- ✅ Block specific dates (holidays, personal days)
- ✅ Override schedule for specific dates
- ✅ Automatic duty status based on schedule
- ✅ Calendar view in nurse app

### 3. FCM Push Notifications

#### Overview
Firebase Cloud Messaging integration for mobile push notifications.

#### Database Schema

```sql
-- Migration: V13__add_fcm_tokens.sql

CREATE TABLE fcm_tokens (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id BIGINT NOT NULL,
    device_token VARCHAR(255) NOT NULL,
    device_type VARCHAR(20) NOT NULL, -- 'ANDROID', 'IOS'
    device_name VARCHAR(100),
    is_active BOOLEAN DEFAULT TRUE,
    last_used_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT fk_fcm_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    CONSTRAINT uk_device_token UNIQUE (device_token)
);

CREATE TABLE notification_preferences (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id BIGINT NOT NULL UNIQUE,
    booking_alerts BOOLEAN DEFAULT TRUE,
    payment_alerts BOOLEAN DEFAULT TRUE,
    review_alerts BOOLEAN DEFAULT TRUE,
    promotional_alerts BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT fk_notif_pref_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE INDEX idx_fcm_tokens_user ON fcm_tokens(user_id);
```

#### Dependencies

### 7. Referral System

#### Database Schema

```sql
-- Migration: V16__add_referral_system.sql

CREATE TABLE referral_codes (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id BIGINT NOT NULL UNIQUE,
    referral_code VARCHAR(20) UNIQUE NOT NULL,
    total_referrals INT DEFAULT 0,
    total_earnings DECIMAL(10, 2) DEFAULT 0.00,
    created_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT fk_referral_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE referrals (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    referrer_user_id BIGINT NOT NULL,
    referred_user_id BIGINT NOT NULL,
    referral_code VARCHAR(20) NOT NULL,
    status VARCHAR(20) DEFAULT 'PENDING', -- PENDING, COMPLETED, REWARDED
    reward_amount DECIMAL(10, 2),
    first_booking_id BIGINT,
    created_at TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP WITHOUT TIME ZONE,
    
    CONSTRAINT fk_referral_referrer FOREIGN KEY (referrer_user_id) REFERENCES users(id),
    CONSTRAINT fk_referral_referred FOREIGN KEY (referred_user_id) REFERENCES users(id),
    CONSTRAINT fk_referral_booking FOREIGN KEY (first_booking_id) REFERENCES bookings(id)
);

CREATE INDEX idx_referrals_referrer ON referrals(referrer_user_id);
CREATE INDEX idx_referrals_code ON referrals(referral_code);
```

#### API Endpoints

